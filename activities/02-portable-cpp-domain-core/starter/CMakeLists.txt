cmake_minimum_required(VERSION 3.24)
project(BudgetBuddyDomain LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(DomainCore STATIC src/BudgetPolicy.cpp)
target_include_directories(DomainCore PUBLIC include)
target_compile_options(DomainCore PRIVATE -Wall -Wextra -Werror)
add_executable(DomainCoreTests tests/BudgetPolicyTests.cpp)
target_link_libraries(DomainCoreTests PRIVATE DomainCore)
add_test(NAME DomainCoreTests COMMAND DomainCoreTests)
